import { useAnimations, useGLTF } from '@react-three/drei'
import bird from '../assets/3d/bird.glb';
import { useEffect, useRef } from 'react';
import { useFrame } from '@react-three/fiber';

function Phoenix({isRotating,...props}) {


  const birdMeshRef = useRef();
  const { scene, animations } = useGLTF(bird);
  const { actions } = useAnimations(animations, birdMeshRef);

  useEffect( () => {
    actions["Take 001"]?.reset().fadeIn(0.5).play();
  }, [actions]);

  useFrame(({ clock }) => {
    const t = clock.elapsedTime;
    const radius = 8;
    const speed = 0.9;

    const x = Math.sin(t * speed) * radius;
    const z = Math.cos(t * speed) * radius - 10;

    const nextX = Math.sin((t + 0.01) * speed) * radius;
    const nextZ = Math.cos((t + 0.01) * speed) * radius - 10;

    birdMeshRef.current.position.x = x;
    birdMeshRef.current.position.z = z;
    birdMeshRef.current.position.y = Math.sin(t * 0.8) * 0.5 + 2;

    const angle = Math.atan2(nextX - x, nextZ - z) + Math.PI / 2 + Math.PI;
    if(x < -7) console.log(`angle: ${angle.toFixed(2)}, x: ${x.toFixed(2)}`);
    birdMeshRef.current.rotation.y = angle;
  })

  return (
    <mesh
    ref={birdMeshRef}
      // position={[-2, -1, 2]}
      scale={[0.004, 0.004, 0.004  ]}
    >
      <primitive object={scene}></primitive>
    </mesh>
  )
}

export default Phoenix
